cmake_minimum_required(VERSION 3.16)
project(odysseus_kde_thumbnail_adapter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui)
find_package(KF6KIO REQUIRED)
add_executable(odysseus-kde-thumbnail-adapter kde-thumbnail-adapter.cpp)
target_link_libraries(odysseus-kde-thumbnail-adapter PRIVATE Qt6::Core Qt6::Gui KF6::KIOCore KF6::KIOGui)
install(TARGETS odysseus-kde-thumbnail-adapter RUNTIME DESTINATION bin)
